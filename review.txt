// =============================================================
//  chat_runtime.js — Chat lifecycle manager (init / cleanup)
//
//  هر بار که صفحه چت جدید swap میشه:
//    1. RtChat.cleanup()  — همه listenerها/observer/timer پاک میشن
//    2. RtChat.init()     — اسکریپت‌های inline اجرا میشن
//
//  اسکریپت‌های inline چت از _rtchatGen برای تشخیص stale بودن
//  استفاده می‌کنن: اگه generation عوض شده باشه، handler skip میشه.
// =============================================================

window.RtChat = (function () {
  'use strict';

  var _generation = 0;
  var _initialized = false;

  // ---- Registries for cleanup ----
  var _docListeners = [];
  var _bodyListeners = [];
  var _observers = [];
  var _timers = [];

  // ==========================
  // Tracking helpers
  // ==========================

  /**
   * addEventListener روی document — قابل cleanup
   */
  function docOn(event, handler, options) {
    var opts = options || false;
    document.addEventListener(event, handler, opts);
    _docListeners.push({ e: event, h: handler, o: opts });
  }

  /**
   * addEventListener روی document.body — قابل cleanup
   */
  function bodyOn(event, handler, options) {
    var opts = options || false;
    document.body.addEventListener(event, handler, opts);
    _bodyListeners.push({ e: event, h: handler, o: opts });
  }

  function trackObserver(obs) {
    if (obs) _observers.push(obs);
    return obs;
  }

  function trackInterval(id) {
    _timers.push({ t: 'i', id: id });
    return id;
  }

  function trackTimeout(id) {
    _timers.push({ t: 't', id: id });
    return id;
  }

  // ==========================
  // CLEANUP
  // ==========================
  function cleanup() {
    if (!_initialized) {
      // حتی اگه init نشده، generation رو بالا ببر
      // تا handlerهای stale قبلی skip بشن
      _generation++;
      window._rtchatGen = _generation;
      return;
    }
    _initialized = false;
    _generation++;
    window._rtchatGen = _generation;

    // 1. document listeners
    var i;
    for (i = 0; i < _docListeners.length; i++) {
      try {
        document.removeEventListener(_docListeners[i].e, _docListeners[i].h, _docListeners[i].o);
      } catch (e) { }
    }
    _docListeners = [];

    // 2. body listeners
    for (i = 0; i < _bodyListeners.length; i++) {
      try {
        document.body.removeEventListener(_bodyListeners[i].e, _bodyListeners[i].h, _bodyListeners[i].o);
      } catch (e) { }
    }
    _bodyListeners = [];

    // 3. observers
    for (i = 0; i < _observers.length; i++) {
      try { _observers[i].disconnect(); } catch (e) { }
    }
    _observers = [];

    // 4. timers
    for (i = 0; i < _timers.length; i++) {
      try {
        if (_timers[i].t === 'i') clearInterval(_timers[i].id);
        else clearTimeout(_timers[i].id);
      } catch (e) { }
    }
    _timers = [];

    // 5. WebSocket close
    var wsForm = document.getElementById('chat_message_form');
    if (wsForm) {
      // HTMX ws extension stores socket in various places
      var sock = wsForm.__htmx_webSocket || wsForm.__htmx_ws;
      if (sock) {
        try { sock.close(); } catch (e) { }
      }
      // Also try htmx internal
      try {
        var api = htmx._('getInternalData');
        if (api) {
          var d = api(wsForm);
          if (d && d.webSocket) {
            d.webSocket.close();
            d.webSocket = null;
          }
        }
      } catch (e) { }
    }

    // 6. Audio stop
    var container = document.getElementById('tg-chat-content');
    if (container) {
      var audios = container.querySelectorAll('audio');
      for (i = 0; i < audios.length; i++) {
        try {
          audios[i].pause();
          audios[i].removeAttribute('src');
          audios[i].load();
        } catch (e) { }
      }
    }

    // 7. MediaRecorder
    if (window._rtchatMediaRecorder) {
      try {
        if (window._rtchatMediaRecorder.state !== 'inactive') {
          window._rtchatMediaRecorder.stop();
        }
      } catch (e) { }
      window._rtchatMediaRecorder = null;
    }

    // 8. Upload XHR
    if (window.rtchatCurrentUploadXhr) {
      try { window.rtchatCurrentUploadXhr.abort(); } catch (e) { }
      delete window.rtchatCurrentUploadXhr;
    }

    // 9. Portaled elements (appended to body by chat.html)
    var portalIds = ['message_action_menu', 'forward_modal', 'file_viewer_modal'];
    for (i = 0; i < portalIds.length; i++) {
      var el = document.getElementById(portalIds[i]);
      if (el && el.dataset.portaled === '1') {
        try { el.remove(); } catch (e) { }
      }
    }

    // 10. Guards reset — so IIFEs can re-bind
    try { delete document.body.dataset.rtchatMsgMenuBound; } catch (e) { }

    // 11. Global function cleanup
    delete window.rtchatClearReply;
    delete window.rtchatMarkOwnMessageScrollPending;
    delete window.rtchatEnsureDateSeparators;
    delete window.rtchatInitLazyImages;
  }

  // ==========================
  // INIT
  // ==========================
  function init() {
    cleanup();

    // Check if we're on a chat page
    var chatContainer = document.getElementById('chat_container');
    if (!chatContainer) return;

    _initialized = true;

    // Execute inline scripts inside the swapped content
    _executeInlineScripts();

    // Scroll to bottom after scripts have run
    if (typeof window.scrollToBottom === 'function') {
      window.scrollToBottom(80, 8);
    }

    // Process any htmx attributes in new content
    var content = document.getElementById('tg-chat-content');
    if (content && window.htmx && typeof window.htmx.process === 'function') {
      window.htmx.process(content);
    }

    // Re-init lazy images if function exists
    if (typeof window.rtchatInitLazyImages === 'function') {
      window.rtchatInitLazyImages(content || document);
    }
  }

  function _executeInlineScripts() {
    var content = document.getElementById('tg-chat-content');
    if (!content) return;

    var scripts = content.querySelectorAll('script');
    for (var i = 0; i < scripts.length; i++) {
      var old = scripts[i];
      var next = document.createElement('script');

      if (old.src) {
        // External script — just re-add
        next.src = old.src;
      } else {
        // Inline script — execute
        next.textContent = old.textContent;
      }

      // Copy attributes (except src which is already handled)
      for (var j = 0; j < old.attributes.length; j++) {
        var attr = old.attributes[j];
        if (attr.name !== 'src') {
          try { next.setAttribute(attr.name, attr.value); } catch (e) { }
        }
      }

      old.parentNode.replaceChild(next, old);
    }
  }

  // ==========================
  // Stale check for IIFEs
  // ==========================

  /**
   * Returns a function that checks if the current generation
   * has changed since capture time. Used by chat.html IIFEs.
   *
   * Usage in IIFE:
   *   var isStale = RtChat.staleGuard();
   *   someElement.addEventListener('click', function(e) {
   *     if (isStale()) return;
   *     // ... handler code
   *   });
   */
  function staleGuard() {
    var myGen = _generation;
    return function () {
      return window._rtchatGen !== myGen;
    };
  }

  // ==========================
  // Public API
  // ==========================
  return {
    init: init,
    cleanup: cleanup,
    staleGuard: staleGuard,
    docOn: docOn,
    bodyOn: bodyOn,
    trackObserver: trackObserver,
    trackInterval: trackInterval,
    trackTimeout: trackTimeout,
    generation: function () { return _generation; },
    isInitialized: function () { return _initialized; }
  };
})();

// Generation marker — initial value
window._rtchatGen = 0;

def chat_view(request, chatroom_identifier='public_chat'):
    if request.path == '/':
        my_groups_qs = request.user.chat_groups.all()
        if not my_groups_qs.exists():
            return render(request, 'layouts/telegram.html')
        last_state = (
            ChatState.objects
            .filter(user=request.user, group__in=my_groups_qs)
            .select_related('group')
            .order_by('-last_read')
            .first()
        )
        if last_state and last_state.group_id:
            chat_group = last_state.group
        else:
            last_group = my_groups_qs.order_by('-id').first()
            if last_group:
                chat_group = last_group
            else:
                chat_group = get_chat_group_by_identifier('public_chat')
    else:
        chat_group = get_chat_group_by_identifier(chatroom_identifier)

    if chat_group.group_name == "public_chat" and not _public_chat_visible_to_user(request.user, chat_group):
        raise Http404

    chatroom_identifier = chat_group.group_slug or chat_group.group_name
    _MSG_RELATED = (
        "author", "author__profile",
        "reply_to", "reply_to__author", "reply_to__author__profile",
        "forwarded_from", "forwarded_from__profile",
    )
    focus_raw = (request.GET.get("focus") or "").strip()
    focus_message = None
    if focus_raw.isdigit():
        focus_message = (
            GroupMessage.objects
            .filter(group=chat_group, id=int(focus_raw))
            .select_related("group")
            .first()
        )

    if focus_message:
        older = list(
            chat_group.chat_messages
            .filter(created__lt=focus_message.created)
            .select_related(*_MSG_RELATED)
            .order_by("-created")[:15]
        )
        newer = list(
            chat_group.chat_messages
            .filter(created__gte=focus_message.created)
            .select_related(*_MSG_RELATED)
            .order_by("created")[:15]
        )
        chat_messages = older[::-1] + newer
    else:
        chat_messages = list(
            chat_group.chat_messages
            .select_related(*_MSG_RELATED)
            .order_by("-created")[:30]
        )[::-1]
    form = ChatmessageCreateForm()

    other_user = None
    if chat_group.is_private:
        if request.user not in chat_group.members.all():
            raise Http404
        for member in chat_group.members.all():
            if member != request.user:
                other_user = member
                break
    private_other_last_read = None
    if chat_group.is_private and other_user:
        other_state = (
            ChatState.objects
            .filter(user=other_user, group=chat_group)
            .only("last_read")
            .first()
        )
        private_other_last_read = other_state.last_read if other_state else None

    if chat_group.groupchat_name and chat_group.group_name != 'public_chat':

        if request.user not in chat_group.members.all():
            raise Http404

            message.group = chat_group
            message.author = request.user
            reply_to_id = (request.POST.get('reply_to') or '').strip()
            if reply_to_id.isdigit():
                message.reply_to = (
                    GroupMessage.objects
                    .filter(group=chat_group, id=int(reply_to_id))
                    .first()
                )
            message.save()
            transaction.on_commit(lambda: _send_push_notifications_for_message(message.id))

            channel_layer = get_channel_layer()
            async_to_sync(channel_layer.group_send)(
                chat_group.group_name,
                {"type": "message_handler", "message_id": message.id},
            )

            target_user_ids = list(chat_group.members.values_list("id", flat=True))
            if (
                not target_user_ids
                and chat_group.group_name == "public_chat"
                and bool(getattr(settings, "CHAT_PUBLIC_CHAT_VISIBLE_TO_ALL", False))
            ):
                refresh_event = {"type": "online_status_handler"}
            else:
                refresh_event = {"type": "online_status_handler", "target_user_ids": (target_user_ids or [request.user.id])}
            async_to_sync(channel_layer.group_send)("online-status", refresh_event)

            context = {
                'message': message,
                'user': request.user,
                'chat_group': chat_group,
                'is_group_admin': _user_is_chat_group_admin(request.user, chat_group),
                'private_other_last_read': private_other_last_read,
            }
            return render(request, 'a_rtchat/partials/chat_message_p.html', context)

    context = {
        'chat_messages': chat_messages,
        'form': form,
        'other_user': other_user,
        'chatroom_identifier': chatroom_identifier,
        'chatroom_ws_name': chat_group.group_name,

window.RtChat = (function () {
  'use strict';

  var _generation = 0;
  var _initialized = false;

  var _docListeners = [];
  var _bodyListeners = [];
  var _observers = [];
  var _timers = [];

  function docOn(event, handler, options) {
    var opts = options || false;
    document.addEventListener(event, handler, opts);
    _docListeners.push({ e: event, h: handler, o: opts });
  }

  function bodyOn(event, handler, options) {
    var opts = options || false;
    document.body.addEventListener(event, handler, opts);
    _bodyListeners.push({ e: event, h: handler, o: opts });
  }

  function trackObserver(obs) {
    if (obs) _observers.push(obs);
    return obs;
  }

  function trackInterval(id) {
    _timers.push({ t: 'i', id: id });
    return id;
  }

  function trackTimeout(id) {
    _timers.push({ t: 't', id: id });
    return id;
  }

  // ==========================
  // CLEANUP
  // ==========================
  function cleanup() {
    _generation++;
    window._rtchatGen = _generation;

    if (!_initialized) return;
    _initialized = false;

    var i;
    for (i = 0; i < _docListeners.length; i++) {
      try { document.removeEventListener(_docListeners[i].e, _docListeners[i].h, _docListeners[i].o); } catch (e) {}
    }
    _docListeners = [];

    for (i = 0; i < _bodyListeners.length; i++) {
      try { document.body.removeEventListener(_bodyListeners[i].e, _bodyListeners[i].h, _bodyListeners[i].o); } catch (e) {}
    }
    _bodyListeners = [];

    for (i = 0; i < _observers.length; i++) {
      try { _observers[i].disconnect(); } catch (e) {}
    }
    _observers = [];

    for (i = 0; i < _timers.length; i++) {
      try {
        if (_timers[i].t === 'i') clearInterval(_timers[i].id);
        else clearTimeout(_timers[i].id);
      } catch (e) {}
    }
    _timers = [];

    // WebSocket
    var wsForm = document.getElementById('chat_message_form');
    if (wsForm) {
      try {
        var api = window.htmx && window.htmx._('getInternalData');
        if (api) {
          var d = api(wsForm);
          if (d && d.webSocket) { d.webSocket.close(); d.webSocket = null; }
        }
      } catch (e) {}
    }

    // Audio
    var content = document.getElementById('tg-chat-content');
    if (content) {
      var audios = content.querySelectorAll('audio');
      for (i = 0; i < audios.length; i++) {
        try { audios[i].pause(); audios[i].removeAttribute('src'); audios[i].load(); } catch (e) {}
      }
    }

    // MediaRecorder
    if (window._rtchatMediaRecorder) {
      try { if (window._rtchatMediaRecorder.state !== 'inactive') window._rtchatMediaRecorder.stop(); } catch (e) {}
      window._rtchatMediaRecorder = null;
    }

    // Upload XHR
    if (window.rtchatCurrentUploadXhr) {
      try { window.rtchatCurrentUploadXhr.abort(); } catch (e) {}
      delete window.rtchatCurrentUploadXhr;
    }

    // Portaled elements
    var portalIds = ['message_action_menu', 'forward_modal', 'file_viewer_modal'];
    for (i = 0; i < portalIds.length; i++) {
      var el = document.getElementById(portalIds[i]);
      if (el && el.dataset.portaled === '1') {
        try { el.remove(); } catch (e) {}
      }
    }

    // Guards reset
    try { delete document.body.dataset.rtchatMsgMenuBound; } catch (e) {}

    // Global functions
    delete window.rtchatClearReply;
    delete window.rtchatMarkOwnMessageScrollPending;
    delete window.rtchatEnsureDateSeparators;
    delete window.rtchatInitLazyImages;
  }

  // ==========================
  // INIT
  // ==========================
  function init() {
    cleanup();

    var chatContainer = document.getElementById('chat_container');
    if (!chatContainer) return;

    _initialized = true;

    // ★ htmx خودش script ها رو اجرا میکنه
    // ما فقط htmx attributes رو process میکنیم
    var content = document.getElementById('tg-chat-content');
    if (content && window.htmx && typeof window.htmx.process === 'function') {
      window.htmx.process(content);
    }

    // اسکرول بعد از اینکه اسکریپت‌ها اجرا شدن
    setTimeout(function() {
      if (typeof window.scrollToBottom === 'function') {
        window.scrollToBottom(0, 8);
      }
    }, 50);
  }

  function staleGuard() {
    var myGen = _generation;
    return function () {
      return window._rtchatGen !== myGen;
    };
  }

  return {
    init: init,
    cleanup: cleanup,
    staleGuard: staleGuard,
    docOn: docOn,
    bodyOn: bodyOn,
    trackObserver: trackObserver,
    trackInterval: trackInterval,
    trackTimeout: trackTimeout,
    generation: function () { return _generation; },
    isInitialized: function () { return _initialized; }
  };
