تغییر ۱: localStorage.clear() تنظیمات کاربر رو پاک نکنه
الان هر بار ورژن عوض بشه، theme و sidebarTab و همه چیز پاک میشه. این رو عوض کن:

تابع cleanupSiteDataIfNeeded رو کامل با این جایگزین کن:

JavaScript

async function cleanupSiteDataIfNeeded() {
  if (sessionStorage.getItem("pwaCleanupPending") !== "1") return;
  sessionStorage.removeItem("pwaCleanupPending");

  var isFullReset = RESET_REQUIRED_DEFAULT;

  // ═══ تنظیمات کاربر رو نگه دار (فقط وقتی reset کامل نیست) ═══
  var preserve = {};
  if (!isFullReset) {
    var keepKeys = ['theme', 'sidebarTab', 'lastChatUrl'];
    try {
      keepKeys.forEach(function (k) {
        var v = localStorage.getItem(k);
        if (v !== null) preserve[k] = v;
      });
    } catch (e) {}
  }

  // ═══ پاکسازی SW ═══
  try {
    if ("serviceWorker" in navigator && navigator.serviceWorker.getRegistrations) {
      var regs = await navigator.serviceWorker.getRegistrations();
      await Promise.all((regs || []).map(function (r) { return r && r.unregister ? r.unregister() : false; }));
    }
  } catch (e) {}

  // ═══ پاکسازی Cache API ═══
  try {
    if ("caches" in window) {
      var names = await caches.keys();
      await Promise.all(names.map(function (n) { return caches.delete(n); }));
    }
  } catch (e) {}

  // ═══ پاکسازی IndexedDB ═══
  if (isFullReset) {
    try {
      if (typeof indexedDB !== "undefined") {
        var dbs = indexedDB.databases ? await indexedDB.databases() : [];
        await Promise.all(
          (dbs || [])
            .map(function (db) { return db && db.name ? db.name : ""; })
            .filter(Boolean)
            .map(function (name) {
              return new Promise(function (resolve) {
                var req = indexedDB.deleteDatabase(name);
                req.onsuccess = function () { resolve(true); };
                req.onerror = function () { resolve(false); };
                req.onblocked = function () { resolve(false); };
              });
            })
        );
      }
    } catch (e) {}
  }

  // ═══ پاکسازی localStorage ═══
  try { localStorage.clear(); } catch (e) {}

  // ═══ تنظیمات رو برگردون ═══
  try {
    Object.keys(preserve).forEach(function (k) {
      localStorage.setItem(k, preserve[k]);
    });
    localStorage.setItem(APP_VERSION_KEY, APP_VERSION);
  } catch (e) {}

  try { window.location.reload(); } catch (e) {}
}
تغییر ۲: cleanup فقط وقتی APP_RESET_REQUIRED هست اجرا بشه
الان با هر تغییر ورژن cleanup اجرا میشه. اما ?v= مشکل cache رو حل میکنه، پس cleanup فقط وقتی واقعاً لازمه باید اجرا بشه:

IIFE مربوط به version check رو پیدا کن و با این جایگزین کن:

JavaScript

(function () {
  try {
    var prev = localStorage.getItem(APP_VERSION_KEY) || "";
    if (prev && prev !== APP_VERSION) {
      sessionStorage.setItem("pwaVersionChanged", "1");
      if (window._dlog) window._dlog('Version changed: ' + prev + ' -> ' + APP_VERSION);
      // فقط وقتی reset کامل لازمه، cleanup کن
      if (RESET_REQUIRED_DEFAULT) {
        sessionStorage.setItem("pwaCleanupPending", "1");
      }
    } else {
      sessionStorage.removeItem("pwaVersionChanged");
    }
    localStorage.setItem(APP_VERSION_KEY, APP_VERSION);
  } catch (e) {
  }
})();
تغییر ۳: iOS flow رو ساده‌تر کن در runFullUpdate
تابع runFullUpdate رو کامل با این جایگزین کن:

JavaScript

async function runFullUpdate() {
  if (!canShowPwaUpdateModal()) {
    markPwaUpdatePending();
    return;
  }
  if (pwaFlowRunning) return;
  pwaFlowRunning = true;

  try {
    // ═══ اگه SW waiting داریم (دسکتاپ/اندروید) ═══
    if (pwaWaitingWorker) {
      pwaStatusText = "";
      pwaStatusKind = "info";
      showUpdateModal();

      if (!pwaPreparedOk) {
        var ok = await prepareUpdate();
        if (!ok) return;
      }
      await applyUpdate();
      return;
    }

    // ═══ بقیه حالات (iOS یا وقتی waiting نیست) ═══
    pwaStatusText = "در حال بروزرسانی...";
    pwaStatusKind = "info";
    showUpdateModal();

    // SW رو unregister و cache رو پاک کن
    try {
      if ("serviceWorker" in navigator && navigator.serviceWorker.getRegistrations) {
        var regs = await navigator.serviceWorker.getRegistrations();
        await Promise.all((regs || []).map(function (r) { return r && r.unregister ? r.unregister() : null; }));
      }
    } catch (e) {}

    try {
      if ("caches" in window) {
        var cacheNames = await caches.keys();
        await Promise.all(cacheNames.map(function (n) { return caches.delete(n); }));
      }
    } catch (e) {}

    // reload
    try {
      window.location.reload(true);
    } catch (e) {
      window.location.reload();
    }
  } finally {
    pwaFlowRunning = false;
    var modal = document.getElementById("pwa-update-modal");
    if (modal && modal.style && modal.style.display !== "none") renderUpdateModal();
  }
}
