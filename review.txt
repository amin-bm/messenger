۲. تابع scrollToBottom — اضافه شدن گارد بیشتر:
JavaScript

function scrollToBottom(time = 0, passes = 5) {
    const delay = Math.max(0, Number(time || 0));
    const maxPasses = Math.max(1, Number(passes || 1));

    setTimeout(() => {
      if (window._rtchatLoadOlderActive) return;
      if (window._rtchatSuppressScroll) return;               // ← NEW
      const fileModal = document.getElementById("file_viewer_modal");
      if (fileModal && !fileModal.classList.contains("hidden")) return;
      if (Date.now() - _fileViewerClosedAt < 3000) return;

      const container = document.getElementById("chat_container");
      if (!container) return;

      let remaining = maxPasses;
      let lastHeight = -1;

      const step = () => {
        if (!container) return;
        if (window._rtchatLoadOlderActive) return;
        if (window._rtchatSuppressScroll) return;              // ← NEW
        const fm = document.getElementById("file_viewer_modal");
        if (fm && !fm.classList.contains("hidden")) return;
        if (Date.now() - _fileViewerClosedAt < 3000) return;
        const nextHeight = container.scrollHeight;
        container.scrollTop = nextHeight;
        remaining -= 1;

        if (remaining <= 0) return;
        if (nextHeight === lastHeight && remaining <= Math.max(0, maxPasses - 2)) return;
        lastHeight = nextHeight;
        requestAnimationFrame(step);
      };

      requestAnimationFrame(step);
    }, delay);
  }

    function scheduleStickScroll(passes = 4) {
      if (!stickToBottom) return;
      if (scheduled) return;
      if (window._rtchatLoadOlderActive) return;

      if (Date.now() - _fileViewerClosedAt < 3000) return;

        scheduled = false;
        if (!stickToBottom) return;
        if (window._rtchatSuppressScroll) return;              // ← NEW

        if (fm && !fm.classList.contains("hidden")) return;
        if (Date.now() - _fileViewerClosedAt < 3000) return;
        scrollToBottom(0, passes);
      });
    }

۵. تابع requestClose در file viewer — کامل جایگزین:
JavaScript

    function requestClose() {
      var chatContainer = document.getElementById("chat_container");
      var savedScrollTop = chatContainer ? chatContainer.scrollTop : null;

      // ── suppress همه اسکرول‌ها ──
      window._rtchatSuppressScroll = true;

      closeUI();

      if (historyPushed) {
        historyPushed = false;
        // به جای history.back() از replaceState استفاده میکنیم
        // تا htmx صفحه رو re-render نکنه
        try {
          var st = Object.assign({}, window.history.state || {});
          delete st.__fileViewerOpen;
          st.__fileViewerClosed = true;
          window.history.replaceState(st, "");
        } catch (e) {}
      }

      // ── حفظ موقعیت اسکرول ──
      if (chatContainer && savedScrollTop !== null) {
        chatContainer.scrollTop = savedScrollTop;
        var restoreCount = 0;
        var restoreScroll = function () {
          if (restoreCount > 20) return;
          restoreCount++;
          var cc = document.getElementById("chat_container");
          if (cc) cc.scrollTop = savedScrollTop;
          requestAnimationFrame(restoreScroll);
        };
        requestAnimationFrame(restoreScroll);
        [50, 150, 300, 500, 750, 1200].forEach(function(ms) {
          setTimeout(function () {
            var cc = document.getElementById("chat_container");
            if (cc) cc.scrollTop = savedScrollTop;
          }, ms);
        });
      }

      // ── recalc stickToBottom و رفع suppress ──
      setTimeout(function() {
        if (typeof window.rtchatForceRecalcSticky === 'function') window.rtchatForceRecalcSticky();
      }, 500);
      setTimeout(function() {
        if (typeof window.rtchatForceRecalcSticky === 'function') window.rtchatForceRecalcSticky();
      }, 1500);
      setTimeout(function() {
        window._rtchatSuppressScroll = false;
        if (typeof window.rtchatForceRecalcSticky === 'function') window.rtchatForceRecalcSticky();
      }, 3100);
    }
